import java.time.Duration;

import org.openqa.selenium.By;
import org.openqa.selenium.JavascriptExecutor;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.support.ui.ExpectedConditions;
import org.openqa.selenium.support.ui.WebDriverWait;

public class Website1 {
	public static void main(String[] args) throws Exception {
		WebDriver driver = new ChromeDriver();
		WebDriverWait wait = new WebDriverWait(driver, Duration.ofSeconds(15));

		try {
			driver.manage().window().maximize();
			driver.get("https://demoqa.com/webtables");

			// 1. Click "Add" button (scroll first to avoid ads blocking the click)
			WebElement addBtn = wait.until(ExpectedConditions.presenceOfElementLocated(By.id("addNewRecordButton")));
			((JavascriptExecutor) driver).executeScript("arguments[0].scrollIntoView({block:'center'});", addBtn);
			wait.until(ExpectedConditions.elementToBeClickable(addBtn)).click();

			// 2. Fill the registration form
			wait.until(ExpectedConditions.visibilityOfElementLocated(By.id("firstName"))).sendKeys("Niyati");
			driver.findElement(By.id("lastName")).sendKeys("Patel");
			driver.findElement(By.id("userEmail")).sendKeys("niyatipatel@gmail.com");
			driver.findElement(By.id("age")).sendKeys("21");
			driver.findElement(By.id("salary")).sendKeys("500000");
			driver.findElement(By.id("department")).sendKeys("MCA");

			// 3. Submit and wait for the form to close
			driver.findElement(By.id("submit")).click();
			wait.until(ExpectedConditions.invisibilityOfElementLocated(By.className("modal-content")));

			// 4. Search the record
			WebElement searchBox = driver.findElement(By.id("searchBox"));
			searchBox.sendKeys("Radhika");

			// 5. Verify the TABLE shows the record
			WebElement tableBody = driver.findElement(By.className("rt-tbody"));
			String tableText = tableBody.getText();

			if (tableText.contains("Radhika") && tableText.contains("amipararadhika@gmail.com")) {
				System.out.println("Record added and searched successfully | PASS");
			} else {
				System.out.println("Record not found in table | FAIL");
			}

			Thread.sleep(3000); // only to see the result in the browser
		} catch (Exception e) {
			System.out.println("Test FAILED due to error: " + e.getMessage());
		} finally {
			driver.quit();
		}
	}
}

public class Website3 {

	static void check(String step, boolean condition) {
		System.out.println((condition ? "PASS : " : "FAIL : ") + step);
	}

	static void clickWhenReady(WebDriverWait wait, By locator) {
		wait.until(ExpectedConditions.elementToBeClickable(locator)).click();
	}

	public static void main(String[] args) throws Exception {

		ChromeOptions options = new ChromeOptions();
		options.addArguments("--disable-notifications");

		WebDriver driver = new ChromeDriver(options);
		WebDriverWait wait = new WebDriverWait(driver, Duration.ofSeconds(15));

		try {
			driver.manage().window().maximize();
			driver.get("https://demowebshop.tricentis.com/");

			// 1. Register (unique email every run)
			clickWhenReady(wait, By.xpath("//a[@href='/register']"));

			String email = "niyati" + System.currentTimeMillis() + "@gmail.com";

			driver.findElement(By.id("ConfirmPassword")).sendKeys("Niyati@12");
			driver.findElement(By.id("register-button")).click();

			String regMsg = wait.until(ExpectedConditions.visibilityOfElementLocated(
					By.cssSelector("div.result"))).getText();
			check("Registration completed", regMsg.contains("registration completed"));

			clickWhenReady(wait, By.className("register-continue-button"));

			// 2. Computers > Desktops
			clickWhenReady(wait, By.xpath("//ul[@class='top-menu']//a[@href='/computers']"));
			clickWhenReady(wait, By.xpath("//div[@class='sub-category-grid']//a[contains(@href,'desktops')]"));
			check("Desktops page opened", driver.getCurrentUrl().contains("desktops"));

			// 3. Open product and add to cart
			clickWhenReady(wait, By.xpath("//a[@href='/build-your-cheap-own-computer']"));
			clickWhenReady(wait, By.id("add-to-cart-button-72"));

			String note = wait.until(ExpectedConditions.visibilityOfElementLocated(
					By.cssSelector("#bar-notification .content"))).getText();
			check("Product added to cart", note.contains("added to your shopping cart"));

			// 4. Cart > terms > checkout
			clickWhenReady(wait, By.cssSelector("a.ico-cart"));
			clickWhenReady(wait, By.id("termsofservice"));
			clickWhenReady(wait, By.id("checkout"));

			// 5. Billing address
			wait.until(ExpectedConditions.visibilityOfElementLocated(By.id("BillingNewAddress_Company")))
					.sendKeys("TCS");
			new Select(driver.findElement(By.id("BillingNewAddress_CountryId"))).selectByVisibleText("India");
			driver.findElement(By.id("BillingNewAddress_City")).sendKeys("Ahmedabad");
			driver.findElement(By.id("BillingNewAddress_Address1")).sendKeys("Vaishnodevi");
			driver.findElement(By.id("BillingNewAddress_ZipPostalCode")).sendKeys("980001");
			driver.findElement(By.id("BillingNewAddress_PhoneNumber")).sendKeys("9876543210");
			clickWhenReady(wait, By.cssSelector("input[onclick='Billing.save()']"));

			// 6. Shipping address (skipped by the site when "ship to same address" is ticked)
			By shipAddr = By.cssSelector("input[onclick='Shipping.save()']");
			By shipMethod = By.cssSelector("input[onclick='ShippingMethod.save()']");
			wait.until(ExpectedConditions.or(
					ExpectedConditions.elementToBeClickable(shipAddr),
					ExpectedConditions.elementToBeClickable(shipMethod)));
			List<WebElement> shipBtn = driver.findElements(shipAddr);
			if (!shipBtn.isEmpty() && shipBtn.get(0).isDisplayed()) {
				shipBtn.get(0).click();
			}

			// 7. Shipping method, payment method, payment info, confirm
			clickWhenReady(wait, shipMethod);
			clickWhenReady(wait, By.cssSelector("input[onclick='PaymentMethod.save()']"));
			clickWhenReady(wait, By.cssSelector("input[onclick='PaymentInfo.save()']"));
			clickWhenReady(wait, By.cssSelector("input[onclick='ConfirmOrder.save()']"));

			// 8. Thank you page
			String msg = wait.until(ExpectedConditions.visibilityOfElementLocated(
					By.tagName("h1"))).getText();
			check("Thank you page shown", msg.equalsIgnoreCase("Thank you"));

			String done = driver.findElement(By.cssSelector("div.order-completed")).getText();
			check("Order processed successfully", done.contains("successfully processed"));

			Thread.sleep(3000); // only to see the result before the browser closes

		} catch (Exception e) {
			System.out.println("FAIL : Test stopped due to error: " + e.getMessage());
		} finally {
			driver.quit();
		}
	}
}

        System.out.println("\n>>> Have aagal: " + nextStep + "  (Enter dabavo)");
        sc.nextLine();

        ChromeOptions options = new ChromeOptions();

        String baseUrl = "https://www.saucedemo.com/";

        driver.get(baseUrl);

        driver.findElement(By.id("password")).sendKeys("secret_sauce");

        error = driver.findElement(By.cssSelector("[data-test='error']")).getText();

        driver.get(baseUrl);
        driver.findElement(By.id("user-name")).sendKeys("standard_user");
        driver.findElement(By.id("password")).sendKeys("wrong_password");
        driver.findElement(By.id("login-button")).click();
        error = driver.findElement(By.cssSelector("[data-test='error']")).getText();

import java.io.File;
import java.io.FileWriter;
import java.time.Duration;
import java.util.List;

import org.openqa.selenium.Alert;
import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.chrome.ChromeOptions;
import org.openqa.selenium.support.ui.ExpectedConditions;
import org.openqa.selenium.support.ui.Select;
import org.openqa.selenium.support.ui.WebDriverWait;

public class TheInternetTest {

    static WebDriver driver;
    static WebDriverWait wait;
    static final String BASE = "https://the-internet.herokuapp.com";

    static void check(String step, boolean condition) {
        System.out.println((condition ? "PASS : " : "FAIL : ") + step);
    }

    static void pause() throws InterruptedException {
        Thread.sleep(1500);
    }

    static void click(By locator) {
        wait.until(ExpectedConditions.elementToBeClickable(locator)).click();
    }

    static void type(By locator, String text) {
        WebElement el = wait.until(ExpectedConditions.visibilityOfElementLocated(locator));
        el.clear();
        el.sendKeys(text);
    }

    static String text(By locator) {
        return wait.until(ExpectedConditions.visibilityOfElementLocated(locator)).getText();
    }

    public static void main(String[] args) throws Exception {

        ChromeOptions options = new ChromeOptions();
        options.addArguments("--disable-notifications");

        driver = new ChromeDriver(options);
        wait = new WebDriverWait(driver, Duration.ofSeconds(20));
        driver.manage().window().maximize();

        try {
            // ---------- 1. Login ----------
            driver.get(BASE + "/login");
            check("Login page opened", driver.getCurrentUrl().contains("/login"));

            // wrong password first
            type(By.id("username"), "tomsmith");
            type(By.id("password"), "wrong_password");
            click(By.cssSelector("button.radius"));
            check("Invalid password message", text(By.id("flash")).contains("Your password is invalid!"));
            pause();

            // correct login
            type(By.id("username"), "tomsmith");
            type(By.id("password"), "SuperSecretPassword!");
            click(By.cssSelector("button.radius"));
            check("Login successful", text(By.id("flash")).contains("You logged into a secure area!"));
            check("Secure area URL", driver.getCurrentUrl().contains("/secure"));
            pause();

            click(By.cssSelector("a[href='/logout']"));
            check("Logout successful", text(By.id("flash")).contains("You logged out of the secure area!"));
            pause();

            // ---------- 2. Checkboxes ----------
            driver.get(BASE + "/checkboxes");
            List<WebElement> boxes = driver.findElements(By.cssSelector("#checkboxes input"));
            check("Checkbox 1 is unchecked at start", !boxes.get(0).isSelected());
            check("Checkbox 2 is checked at start", boxes.get(1).isSelected());
            boxes.get(0).click();
            boxes.get(1).click();
            check("Checkbox 1 is checked after click", boxes.get(0).isSelected());
            check("Checkbox 2 is unchecked after click", !boxes.get(1).isSelected());
            pause();

            // ---------- 3. Dropdown ----------
            driver.get(BASE + "/dropdown");
            Select dropdown = new Select(wait.until(ExpectedConditions.visibilityOfElementLocated(By.id("dropdown"))));
            dropdown.selectByVisibleText("Option 2");
            check("Dropdown shows Option 2", dropdown.getFirstSelectedOption().getText().equals("Option 2"));
            pause();

            // ---------- 4. JavaScript alerts ----------
            driver.get(BASE + "/javascript_alerts");

            click(By.cssSelector("button[onclick='jsAlert()']"));
            Alert alert = wait.until(ExpectedConditions.alertIsPresent());
            check("JS Alert text", alert.getText().equals("I am a JS Alert"));
            alert.accept();
            check("Alert result", text(By.id("result")).contains("You successfully clicked an alert"));
            pause();

            click(By.cssSelector("button[onclick='jsConfirm()']"));
            alert = wait.until(ExpectedConditions.alertIsPresent());
            alert.accept();
            check("Confirm OK result", text(By.id("result")).contains("You clicked: Ok"));
            pause();

            click(By.cssSelector("button[onclick='jsConfirm()']"));
            alert = wait.until(ExpectedConditions.alertIsPresent());
            alert.dismiss();
            check("Confirm Cancel result", text(By.id("result")).contains("You clicked: Cancel"));
            pause();

            click(By.cssSelector("button[onclick='jsPrompt()']"));
            alert = wait.until(ExpectedConditions.alertIsPresent());
            alert.sendKeys("Radhika");
            alert.accept();
            check("Prompt result", text(By.id("result")).contains("You entered: Radhika"));
            pause();

            // ---------- 5. File upload ----------
            File file = File.createTempFile("upload_test", ".txt");
            file.deleteOnExit();
            try (FileWriter fw = new FileWriter(file)) {
                fw.write("Selenium file upload test");
            }

            driver.get(BASE + "/upload");
            driver.findElement(By.id("file-upload")).sendKeys(file.getAbsolutePath());
            click(By.id("file-submit"));
            check("File Uploaded heading", text(By.tagName("h3")).equals("File Uploaded!"));
            check("Uploaded file name shown", text(By.id("uploaded-files")).equals(file.getName()));

            Thread.sleep(3000);

        } catch (Exception e) {
            System.out.println("FAIL : Test stopped due to error: " + e.getMessage());
        } finally {
            driver.quit();
        }
    }
}

import org.openqa.selenium.support.ui.ExpectedConditions;
import org.openqa.selenium.support.ui.FluentWait;
import org.openqa.selenium.support.ui.WebDriverWait;

public class UITestingPlaygroundTest {

    static WebDriver driver;
    static WebDriverWait wait;
    static final String BASE = "http://uitestingplayground.com";

    // ===== CHANGE THE SPEED HERE (milliseconds, 1000 = 1 second) =====
    static final int ACTION_DELAY = 1500;
    static final int STEP_DELAY = 3000;

    interface Step {
        void run() throws Exception;
    }

    static void check(String step, boolean condition) {
        System.out.println((condition ? "PASS : " : "FAIL : ") + step);
    }

    static void pause() {
        try {
            Thread.sleep(STEP_DELAY);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
    }

    static void slow() {
        try {
            Thread.sleep(ACTION_DELAY);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
    }

    // runs one part of the test; if it fails, the other parts still run
    static void section(String name, Step step) {
        System.out.println("\n--- " + name + " ---");
        try {
            step.run();
        } catch (Exception e) {
            String msg = e.getMessage() == null ? e.toString() : e.getMessage().split("\n")[0];
            System.out.println("FAIL : " + name + " stopped: " + msg);
        }
        pause();
    }

    static WebElement scrollTo(By locator) {
        WebElement el = wait.until(ExpectedConditions.presenceOfElementLocated(locator));
        ((JavascriptExecutor) driver).executeScript("arguments[0].scrollIntoView({block:'center'});", el);
        return el;
    }

    // real mouse click (some pages ignore JavaScript clicks)
    static void click(By locator) {
        scrollTo(locator);
        wait.until(ExpectedConditions.elementToBeClickable(locator)).click();
        slow();
    }

    static void type(By locator, String text) {
        WebElement el = wait.until(ExpectedConditions.visibilityOfElementLocated(locator));
        el.clear();
        el.sendKeys(text);
        slow();
    }

    static String text(By locator) {
        return wait.until(ExpectedConditions.visibilityOfElementLocated(locator)).getText();
    }

    public static void main(String[] args) throws Exception {

        ChromeOptions options = new ChromeOptions();
        options.addArguments("--disable-notifications");

        driver = new ChromeDriver(options);
        wait = new WebDriverWait(driver, Duration.ofSeconds(30));
        driver.manage().window().maximize();

        try {
            // 1. Dynamic ID: the id changes on every page load, so the text is used as locator
            section("Dynamic ID", () -> {
                driver.get(BASE + "/dynamicid");
                By btn = By.xpath("//button[text()='Button with Dynamic ID']");
                String id1 = wait.until(ExpectedConditions.visibilityOfElementLocated(btn)).getAttribute("id");
                driver.navigate().refresh();
                String id2 = wait.until(ExpectedConditions.visibilityOfElementLocated(btn)).getAttribute("id");
                check("ID changes after refresh (so id is not a good locator)", !id1.equals(id2));
                click(btn);
                check("Button found and clicked using its text", true);
            });

            // 2. Class attribute: the button has several classes
            section("Class Attribute", () -> {
                driver.get(BASE + "/classattr");
                click(By.xpath("//button[contains(concat(' ', normalize-space(@class), ' '), ' btn-primary ')]"));
                Alert alert = wait.until(ExpectedConditions.alertIsPresent());
                check("Primary button alert shown", alert.getText().contains("Primary button pressed"));
                alert.accept();
            });

            // 3. Click: the page ignores JavaScript clicks, only a real mouse click works

                FluentWait<WebDriver> fluent = new FluentWait<>(driver)
                        .withTimeout(Duration.ofSeconds(30))

                check("AJAX data loaded", label.getText().contains("Data loaded with AJAX get request."));
            });

            // 6. Client side delay: Explicit Wait
            section("Client Side Delay (Explicit Wait)", () -> {

            });

import java.time.Duration;

import org.openqa.selenium.By;

import org.openqa.selenium.support.ui.ExpectedConditions;
import org.openqa.selenium.support.ui.WebDriverWait;

public class GlobalSQATest {

    static WebDriver driver;
    static WebDriverWait wait;
    static final String BASE = "https://www.globalsqa.com/demo-site/";

    // ===== CHANGE THE SPEED HERE (milliseconds, 1000 = 1 second) =====
    static final int ACTION_DELAY = 1500;
    static final int STEP_DELAY = 3000;

    interface Step {
        void run() throws Exception;
    }

    static void check(String step, boolean condition) {
        System.out.println((condition ? "PASS : " : "FAIL : ") + step);
    }

    static void pause() {
        try {
            Thread.sleep(STEP_DELAY);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
    }

    static void slow() {
        try {
            Thread.sleep(ACTION_DELAY);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
    }

    static void section(String name, Step step) {
        System.out.println("\n--- " + name + " ---");
        try {
            step.run();
        } catch (Exception e) {
            String msg = e.getMessage() == null ? e.toString() : e.getMessage().split("\n")[0];
            System.out.println("FAIL : " + name + " stopped: " + msg);
        }
        driver.switchTo().defaultContent();
        pause();
    }

    static WebElement scrollTo(By locator) {
        WebElement el = wait.until(ExpectedConditions.presenceOfElementLocated(locator));
        ((JavascriptExecutor) driver).executeScript("arguments[0].scrollIntoView({block:'center'});", el);
        return el;
    }

    // the demos are inside an iframe: switch to the first visible demo frame
    static void switchToDemoFrame() {
        wait.until(d -> !d.findElements(By.cssSelector("iframe.demo-frame")).isEmpty());
        List<WebElement> frames = driver.findElements(By.cssSelector("iframe.demo-frame"));
        for (WebElement f : frames) {
            if (f.isDisplayed()) {
                ((JavascriptExecutor) driver).executeScript("arguments[0].scrollIntoView({block:'center'});", f);
                driver.switchTo().frame(f);
                return;
            }
        }
        driver.switchTo().frame(frames.get(0));
    }

    public static void main(String[] args) throws Exception {

        ChromeOptions options = new ChromeOptions();
        options.addArguments("--disable-notifications");

        driver = new ChromeDriver(options);
        wait = new WebDriverWait(driver, Duration.ofSeconds(30));
        driver.manage().window().maximize();

        try {
            // 1. Date picker
            section("Date picker", () -> {
                driver.get(BASE + "datepicker/");
                check("Date picker page opened", driver.getCurrentUrl().contains("datepicker"));
                switchToDemoFrame();
                WebElement input = wait.until(ExpectedConditions.elementToBeClickable(By.id("datepicker")));
                input.click();
                slow();
                wait.until(ExpectedConditions.visibilityOfElementLocated(By.className("ui-datepicker-calendar")));
                driver.findElement(By.xpath("//table[@class='ui-datepicker-calendar']//a[text()='15']")).click();
                slow();
                String value = driver.findElement(By.id("datepicker")).getAttribute("value");
                check("Date selected: " + value, value != null && value.contains("15"));
            });

            // 2. Drag and drop (photo manager: drag a photo to the trash)
            section("Drag and drop", () -> {
                driver.get(BASE + "draganddrop/");
                check("Drag and drop page opened", driver.getCurrentUrl().contains("draganddrop"));
                switchToDemoFrame();
                wait.until(ExpectedConditions.presenceOfElementLocated(By.cssSelector("#gallery li")));
                int before = driver.findElements(By.cssSelector("#gallery li")).size();
                WebElement photo = driver.findElement(By.cssSelector("#gallery li"));
                WebElement trash = driver.findElement(By.id("trash"));
                new Actions(driver).clickAndHold(photo).moveByOffset(10, 10)
                        .moveToElement(trash).pause(Duration.ofMillis(500)).release().perform();
                slow();
                int after = driver.findElements(By.cssSelector("#gallery li")).size();
                check("Photo moved to trash (" + before + " -> " + after + ")", after < before);
            });

            // 3. Tabs
            section("Tabs", () -> {
                driver.get(BASE + "tabs/");
                check("Tabs page opened", driver.getCurrentUrl().contains("tabs"));
                switchToDemoFrame();
                wait.until(ExpectedConditions.elementToBeClickable(By.cssSelector("a[href='#tabs-2']"))).click();
                slow();
                check("Second tab content is visible",
                        wait.until(ExpectedConditions.visibilityOfElementLocated(By.id("tabs-2"))).isDisplayed());
                driver.findElement(By.cssSelector("a[href='#tabs-3']")).click();
                slow();
                check("Third tab content is visible",
                        wait.until(ExpectedConditions.visibilityOfElementLocated(By.id("tabs-3"))).isDisplayed());
            });

            Thread.sleep(5000);

        } finally {
            driver.quit();
        }
    }
